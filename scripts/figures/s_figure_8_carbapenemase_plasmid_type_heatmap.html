get_presence_absence_matrix_sep_specific_split<- function(df,variable,separator){
  presences <- str_split(unlist(c(df %>% dplyr::select(paste0(variable)))),separator) %>% unlist(.)  %>% unique(.) %>% .[!. %in% "-"] %>% sort %>% subset(!. %in% c(NA,"")) 
  vector <- as.vector(tidyr::unite(df %>% dplyr::select(paste0(variable)),determinants,sep =separator)) %>% unlist() %>% as.vector
  results <- matrix(nrow = nrow(df),ncol=length(presences))  %>% `colnames<-`(presences)
for(i in presences){
  results[,paste0(i)] <-  results[,paste0(i)] <-as.vector(ifelse(unlist(lapply(str_split(as.matrix(df[,paste0(variable)]),separator), function(x) i %in% x))==TRUE,1,0))
}
  results <- as.data.frame(results)
  results$isolate_no = df$isolate_no
  results$carbapenemase_plasmid = df$primary_cluster_id
  results$carbapenemase = df$carbapenemase
     
  return(results)
}
