df_carb_omp <- df %>% select(isolate_no,carbapenemase,study,Species_ST_recode) %>% left_join(., carb_omp %>% mutate(isolate_no = rownames(.)))    %>% as.data.frame

# Scales 
## Genotypes
genotypes <- carbapenemase_scale$breaks  %>% gsub("<i>bla</i><sub>","bla",.) %>% gsub("</sub>","",.) %>% trimws(.,"both")
genotype_colors <- setNames(carbapenemase_scale$palette(1), genotypes)
 
## Study coloring
study_colors <-  study_comparison_scale_fill$palette(1)

# Row annotation
legend_common_features <- list(
  title_gp = gpar(fontsize = 9),
  labels_gp = gpar(fontsize = 9),
  grid_width = unit(0.25, "cm"),
  grid_height = unit(0.25, "cm")
)

row_annots <- rowAnnotation(Study = df_carb_omp$study,
                            `Sequence type (ST)` = df_carb_omp$Species_ST_recode,
                            Carbapenemase = df_carb_omp$carbapenemase,
                            col = list(Study = study_colors,
                                       Carbapenemase = genotype_colors,
                                       `Sequence type (ST)`=ST_palette %>% subset(names(.) %in% c(unique(df_carb_omp$Species_ST_recode)))), 
                            annotation_name_gp = gpar(fontsize = 9),
                            annotation_legend_param =list(
    Study = c(legend_common_features,list(ncol=2)), 
    `Sequence type (ST)` =c(legend_common_features,list(at = ST_scale$breaks,ncol=2)),  
    Carbapenemase =c(legend_common_features,list(at=carbapenemase_scale$breaks %>% gsub("<i>bla</i><sub>","bla",.) %>% gsub("</sub>","",.) %>% trimws(.,"both") ,ncol=1)))
 )

# Create genotype matrix
genotype_matrix <- df_carb_omp %>%  select(-carbapenemase,-study,-Species_ST_recode) %>% mutate_all(as.numeric) %>% select_if(colSums(.,na.rm = T)>0) %>% subset(is.na(rowSums(.))==F)
colnames(genotype_matrix) <- recode_genotypes(colnames(genotype_matrix))

# Heatmap
porin_bla_genotype <- ComplexHeatmap::Heatmap(genotype_matrix,
                                              col=c("white","black"),
                                              show_row_names = F,
                                              show_row_dend = T,
                                              show_column_dend = T,
                                              show_parent_dend_line = F,
                                              name = "Genotype",
                                              gap = unit(0.2, "mm") ,
                                              right_annotation = row_annots,
  heatmap_legend_param = list(
    labels = c("1" = "Present","0" = "Absent"),
    nrow = 1,
     gap = unit(3, "mm"),
